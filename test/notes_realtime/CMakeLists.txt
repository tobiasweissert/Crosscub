if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  cmake_minimum_required(VERSION 3.16)
  project(notes_realtime_tests NONE)
  enable_testing()
endif()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME NotesRealtimeNormal COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/run.py)
add_test(NAME NotesRealtimeSanitized COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/run.py --sanitize)
set_tests_properties(NotesRealtimeNormal NotesRealtimeSanitized PROPERTIES TIMEOUT 180 RUN_SERIAL TRUE)
